cmake_minimum_required(VERSION 3.20)
project(NetEngine Version 1.0.0 LANGUAGES CXX)


# Global Standard - C++17 required for std::optional, structured bindings, and inline variables
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Output directories for build artifacts; sets a stable path to grab from
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/lib)

# Options toggled at configure time via -DOPTION=ON/OFF
option(NETENGINE_BUILD_TESTS "Build unit tests (requires GoogleTest)" OFF)
option(NETENGINE_USE_OPENSSL "Use OPENSSL for AES-128/HMAC Crypto" ON)
option(NETENGINE_ENABLE_LOGGING "Enable internal debug logging" ON)

# Platform Socket Libraries
if(WIN32)
    set(PLATFORM_LIBS Ws2_32)
elseif(UNIX)
    set(PLATFORM_LIBS "")
endif()

#OpenSSL Dependency
if(NETENGINE_USE_OPENSSL)
    find_paackage(OpenSSL REQUIRED)
    add_compile_definitions(NETENGINE_USE_OPENSSL)
endif()

# Compiler-wide Definitions
if(NETENGINE_ENABLE_LOGGING)
    add_compile_definitions(NETENGINE_LOGGING)
endif()

# Suppress deprecation warnings on MSVC for POSIX names
if(MSVC)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS _WINSOCK_DEPRECARED_NO_WARNINGS)
    add_compile_options(/W4 /WX- /MP)
else()
    add_compile_options(-Wall -Wextra -Wno-unused-parameter)
endif()



# Gathering all source files from subsystems

#set(NETENGINE_SOURCES
    # Transport Layer
    # Serializer
    # Replication
    # Packet Crypto
    # Genre Modules
    # C ABI Export Layer
#)



# Shared library target
# SHARED keyword causes CMake to build a .dll/.so/.dylib
# The C# DLLimport will load this at runtime
add_library(NetEngine_Core PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/cpp_core
    ${CMAKE_CURRENT_SOURCE_DIR}/cpp_core/Transport
    ${CMAKE_CURRENT_SOURCE_DIR}/cpp_core/Serializer
    ${CMAKE_CURRENT_SOURCE_DIR}/cpp_core/Replication
    ${CMAKE_CURRENT_SOURCE_DIR}/cpp_core/Packets
    ${CMAKE_CURRENT_SOURCE_DIR}/cpp_core/Genres
)

if(NETENGINE_USE_OPENSSL)
    target_link_libraries(NetEngine_Core PRIVATE OpenSSL::Crypto OpenSSL::SSL)
endif()

target_link_libraries(NetEngine_Core PRIVATE ${PLATFORM_LIBS})


# Export all symbols on Windows (needed for DLLimport to resolve)
if(WIN32)
    set_target_properties(NetEngine_Core PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
endif()


##################### Testing #####################

if(NETENGINE_BUILD_TESTS)
    enable_testing()
    add_subdirectory(tests)
endif()

message(STATUS "NetEngine ${PROJECT_VERSION} configured")
message(STATUS "OpenSSL: ${NETENGINE_USE_OPENSSL}")
message(STATUS "Logging: ${NETENGINE_ENABLE_LOGGING}")
message(STATUS "Tests: ${NETENGINE_BUILD_TESTS}")
